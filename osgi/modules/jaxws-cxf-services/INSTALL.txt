# 1. download CXF 3.0-milestone2 from https://cxf.apache.org/download.html
mkdir /tmp/cxf
cd /tmp/cxf
# Czech CDN, may be slow from outside Europe
wget ftp://mirror.hosting90.cz/apache/cxf/3.0.0-milestone2/apache-cxf-3.0.0-milestone2.tar.gz

# 2. Unzip into /tmp/cxf/apache-cxf-3.0.0-milestone2
tar -xvzf apache-cxf-3.0.0-milestone2.tar.gz

# 3. Workaround missing resource bundle: change MANIFEST.MF of wss4j-ws-security-common-2.0.0-rc1.jar << add following Import-Package declaration: org.apache.xml.security.resource;version="[2.0,3)"
unzip apache-cxf-3.0.0-milestone2/lib/wss4j-ws-security-common-2.0.0-rc1.jar META-INF/MANIFEST.MF
sed 's/Import-Package: /Import-Package: org.apache.xml.security.resource;version="[2.0,3)",/' -i META-INF/MANIFEST.MF
zip apache-cxf-3.0.0-milestone2/lib/wss4j-ws-security-common-2.0.0-rc1.jar META-INF/MANIFEST.MF
rm -r META-INF

# 4. Download OSGi-fied version of packaged JARs, save into /tmp/cxf
# asm-3.3.1.jar:
wget http://repo1.maven.org/maven2/org/apache/servicemix/bundles/org.apache.servicemix.bundles.asm/3.3.1_1/org.apache.servicemix.bundles.asm-3.3.1_1.jar -O /tmp/cxf/org.apache.servicemix.bundles.asm-3.3.1_1.jar
# jasypt-1.9.1.jar:
wget http://repo1.maven.org/maven2/org/apache/servicemix/bundles/org.apache.servicemix.bundles.jasypt/1.9.1_2/org.apache.servicemix.bundles.jasypt-1.9.1_2.jar -O /tmp/cxf/org.apache.servicemix.bundles.jasypt-1.9.1_2.jar
# opensaml-2.6.0.jar:
wget http://repo1.maven.org/maven2/org/apache/servicemix/bundles/org.apache.servicemix.bundles.opensaml/2.6.0_3/org.apache.servicemix.bundles.opensaml-2.6.0_3.jar -O /tmp/cxf/org.apache.servicemix.bundles.opensaml-2.6.0_3.jar
# javassist-3.12.1.ga_1.jar
wget http://repo1.maven.org/maven2/org/apache/servicemix/bundles/org.apache.servicemix.bundles.javassist/3.12.1.ga_1/org.apache.servicemix.bundles.javassist-3.12.1.ga_1.jar  -O /tmp/cxf/org.apache.servicemix.bundles.javassist-3.12.1.ga_1.jar

# 5: Initialize keystores in the project
cd PROJECT_DIR... (/opt/liferay.git/portal.git/osgi/modules/jaxws-cxf-services/)
cd keys
./create.sh
cd ..

# 6. Build application
mvn clean package

# 7. Install CXF into OSGi using gogo telnet console using install-osgi.script

# 7.2 - compile and install cxf-rt-ws-policy-patch
cd /opt/liferay.git/portal/osgi/modules/cxf-rt-ws-policy-patch
mvn package
install file:/opt/liferay.git/portal/osgi/modules/cxf-rt-ws-policy-patch/target/cxf-rt-ws-policy-patch-1.0.0.jar
refresh file:/tmp/cxf/apache-cxf-3.0.0-milestone2/lib/cxf-rt-ws-policy-3.0.0-milestone2.jar


# 8. Download nad install commons-lang

wget http://central.maven.org/maven2/commons-lang/commons-lang/2.6/commons-lang-2.6.jar -O /tmp/cxf/commons-lang-2.6.jar
install file:/tmp/cxf/commons-lang-2.6.jar

# 8: Run application

uninstall file:/opt/liferay.git/portal.git/osgi/modules/jaxws-cxf-services/target/jaxws-cxf-services-1.0.0.jar
resolve
refresh
felix:start file:/opt/liferay.git/portal.git/osgi/modules/jaxws-cxf-services/target/jaxws-cxf-services-1.0.0.jar
